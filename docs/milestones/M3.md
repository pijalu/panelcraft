## Frozen FS API (M3-WP1 — M4+ consumes these signatures)

```ts
type NodeKind = 'file' | 'folder';
interface FsNode { kind: NodeKind; name: string; content?: string; children?: FsNode[] }
interface Workspace { name: string; root: FsNode }               // root: folder, name ''
class FsError extends Error { code: FsErrorCode; path: string }  // codes: invalid-path,
// invalid-name, reserved-name, not-found, exists, is-folder, is-file, into-self
const MANIFEST_NAME = 'panelcraft.workspace.json';
newWorkspace(name?): Workspace
splitPath(path): string[]        // '' → []; rejects .. \x00-\x1f \ >255; '.' skipped
joinPath(...parts): string; parentPath(path): string; basename(path): string
getNode(ws, path): FsNode | null
readFile(ws, path): string       // throws not-found / is-folder
createFile(ws, path, content?): FsNode   // mkdirp parents; throws exists/reserved-name
writeFile(ws, path, content): FsNode     // create-or-overwrite
deleteNode(ws, path): void       // recursive; root protected
renameNode(ws, path, newName): void      // unique-in-parent; reserved at root
moveNode(ws, from, toDir): void          // keeps basename; into-own-subtree rejected
duplicateNode(ws, path): string          // deep copy "a copy.pcd" / "a copy 2.pcd"
listDir(ws, path): FsNode[]              // folders first, then files, localeCompare
walkNodes(ws, path?): {path, node}[]     // pre-order, folders first
filesUnder(ws, path?): {path, node}[]    // files only
```
