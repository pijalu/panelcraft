## Frozen FS API (M3-WP1 — M4+ consumes these signatures)

```ts
type NodeKind = 'file' | 'folder';
interface FsNode { kind: NodeKind; name: string; content?: string; children?: FsNode[] }
class FsError extends Error { code: FsErrorCode; path: string }  // codes: invalid-path,
// invalid-name, reserved-name, not-found, exists, is-folder, is-file, into-self
const MANIFEST_NAME = 'panelcraft.workspace.json';
newRoot(): FsNode                // fresh workspace tree; display name lives in app state
splitPath(path): string[]        // '' → []; rejects .. \x00-\x1f \ >255; '.' skipped
joinPath(...parts): string; parentPath(path): string; basename(path): string
getNode(root, path): FsNode | null
readFile(root, path): string     // throws not-found / is-folder
createFile(root, path, content?): FsNode // mkdirp parents; throws exists/reserved-name
writeFile(root, path, content): FsNode   // create-or-overwrite
deleteNode(root, path): void     // recursive; root protected
renameNode(root, path, newName): void    // unique-in-parent; reserved at root
moveNode(root, from, toDir): void        // keeps basename; into-own-subtree rejected
duplicateNode(root, path): string        // deep copy "a copy.pcd" / "a copy 2.pcd"
listDir(root, path): FsNode[]            // folders first, then files, localeCompare
walkNodes(root, path?): {path, node}[]   // pre-order, folders first
filesUnder(root, path?): {path, node}[]  // files only
```
